const game = {
  manifest: {
    id: 'online.score-race', version: '1.0.0', contentVersion: '1.0.0', sdkRange: '^0.1.0',
    name: '夺分赛', description: '轮流抢分，先达到 5 分获胜。在线安装示例。',
    players: { min: 2, max: 4 }, mode: 'rules-driven', capabilities: ['turn-based'],
    defaultAssetPack: { id: 'online.score-race.default', version: '1.0.0' }, developmentOnly: false
  },
  validateOptions(value) {
    if (!value || typeof value !== 'object' || Array.isArray(value) || Object.keys(value).length) throw new Error('No options');
    return {};
  },
  parseAction(value) {
    if (!value || value.type !== 'claim' || Object.keys(value).length !== 1) throw new Error('Invalid action');
    return { type: 'claim' };
  },
  setup({ seats }) {
    return { state: { seats, scores: seats.map(() => 0), turn: 0, winner: null }, events: [] };
  },
  getView(state, viewer) {
    return { seats: state.seats, scores: state.scores, currentSeat: state.seats[state.turn], winner: state.winner, canClaim: state.winner === null && viewer.kind === 'seat' && viewer.seatId === state.seats[state.turn] };
  },
  getActionSpec(state, viewer) {
    return { actions: this.getView(state, viewer).canClaim ? [{ type: 'claim' }] : [] };
  },
  validateAction(state, actor, action) {
    this.parseAction(action);
    if (actor.kind !== 'seat' || actor.seatId !== state.seats[state.turn] || state.winner !== null) throw new Error('Not your turn');
  },
  applyAction(state, actor, action, rng) {
    this.validateAction(state, actor, action);
    const points = rng.nextInt(1, 3);
    state.scores[state.turn] += points;
    if (state.scores[state.turn] >= 5) state.winner = actor.seatId;
    else state.turn = (state.turn + 1) % state.seats.length;
    return { state, events: [{ type: 'claimed', seatId: actor.seatId, points }] };
  },
  projectEvents(events) { return events; },
  getOutcome(state) {
    return state.winner === null ? { status: 'ongoing' } : { status: 'finished', winners: [state.winner], scores: Object.fromEntries(state.seats.map((seat, i) => [seat, state.scores[i]])) };
  },
  serialize(state) { return state; },
  deserialize(value) {
    if (!value || !Array.isArray(value.seats) || value.seats.length < 2 || value.seats.length > 4 || !Array.isArray(value.scores) || value.scores.length !== value.seats.length || !value.scores.every(score => Number.isInteger(score) && score >= 0 && score <= 6) || !Number.isInteger(value.turn) || value.turn < 0 || value.turn >= value.seats.length || (value.winner !== null && !value.seats.includes(value.winner))) throw new Error('Invalid save');
    return value;
  },
  getFallbackAction(view) { return view.canClaim ? { type: 'claim' } : null; }
};
